# Функции: определения_совещания

functions_version: 1
format: 2.22

> Определения пилота «Совещание» (выписка методики по формату 2.1, разметка функций: судьба «Опр», 13 функций, и определения таблицы «Определения» выписки),
> приведённые к формату 2.5. Типы нужны только для подписей определений; запись находок — как в выписке.

## Типы

```
тип Критичность перечисление Критическая | Значимая | Желательная
тип Находка { критичность Критичность, доказательство строка, повтор булево, вопрос булево }
тип Линза { идентификатор строка, потери_подряд число>=0, пробы число>=0, попытка число>=0 }
тип ВидВызывающего перечисление Линза | СессияЦикла | ХодЧата | Вратарь
тип ПределыАгента { срок_тишины ?число, бюджет_токенов ?число, бюджет_стоимости ?вещественное, предел_ходов ?число }
```

## Константы

| имя | тип | значение |
|---|---|---|
| СРОК_БЕЗДЕЙСТВИЯ | число>=0 | 0 |

## Функции

```
функция есть_критические(находки []Находка) -> булево =>
  находки.любой(н => н.критичность == Критичность.Критическая)

функция желательные(находки []Находка) -> []Находка =>
  находки.фильтр(н => н.критичность == Критичность.Желательная)

функция желательные_и_значимые(находки []Находка) -> []Находка =>
  находки.фильтр(н => н.критичность == Критичность.Желательная или н.критичность == Критичность.Значимая)

функция желательные_и_значимые_с_повтором(находки []Находка) -> []Находка =>
  находки.фильтр(н => н.критичность == Критичность.Желательная
                      или (н.критичность == Критичность.Значимая и н.повтор))

функция критические_и_значимые_без_повтора(находки []Находка) -> []Находка =>
  находки.фильтр(н => н.критичность == Критичность.Критическая
                      или (н.критичность == Критичность.Значимая и не н.повтор))

функция не_вошедшие_в(находки []Находка, среди []Находка) -> []Находка =>
  находки.фильтр(н => не среди.любой(с => с.доказательство == н.доказательство))

функция объединить_без_повторов(а []Находка, б []Находка) -> []Находка =>
  а + не_вошедшие_в(б, а)

функция находки_всех_линз(порядок []строка, находки словарь[(строка, число>=0), []Находка], н число>=0) -> []Находка =>
  порядок.свёртка([], (акк, л) => акк + (находки[(л, н)] ?? []))

функция есть_вопросы_проверяющих(порядок []строка, находки словарь[(строка, число>=0), []Находка], н число>=0) -> булево =>
  порядок.любой(л => (находки[(л, н)] ?? []).любой(ф => ф.вопрос))

функция вопросы_проверяющих(порядок []строка, находки словарь[(строка, число>=0), []Находка], н число>=0) -> ?[]Находка {
  пусть вопросы = находки_всех_линз(порядок, находки, н).фильтр(ф => ф.вопрос);
  если вопросы == [] то { пусто } иначе { вопросы }
}

функция линзы_без_критических(порядок []строка, находки словарь[(строка, число>=0), []Находка], н число>=0) -> []строка =>
  порядок.фильтр(л => не есть_критические(находки[(л, н)] ?? []))

функция вид_вызова(сторона строка, линзы словарь[строка, Линза]) -> ВидВызывающего =>
  если есть(линзы[сторона]) то { ВидВызывающего.Линза } иначе { ВидВызывающего.ХодЧата }

функция пределы_роли() -> ПределыАгента =>
  ПределыАгента { срок_тишины: если СРОК_БЕЗДЕЙСТВИЯ == 0 то { пусто } иначе { СРОК_БЕЗДЕЙСТВИЯ },
                  бюджет_токенов: пусто, бюджет_стоимости: пусто, предел_ходов: пусто }

функция слово_да_нет(б булево) -> строка =>
  если б то { "yes" } иначе { "no" }
```
