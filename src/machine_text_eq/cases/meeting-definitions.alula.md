# Определения: определения_совещания

definitions_version: 1
format: 2.14

> Определения пилота «Совещание» (выписка методики по формату 2.1, разметка функций: судьба «Опр», 13 функций, и определения таблицы «Определения» выписки),
> приведённые к формату 2.5. Типы нужны только для подписей определений; запись находок — как в выписке.

## Типы

```
перечисление Критичность (Severity) = Критическая (Critical) | Значимая (Significant) | Желательная (Desirable)
тип Находка (Finding) { критичность (severity) Критичность, доказательство (evidence) текст, повтор (repeat) да/нет, вопрос (question) да/нет }
тип Линза (Lens) { идентификатор (id) текст, потери_подряд (lost) число>=0, пробы (probes) число>=0, попытка (attempt) число>=0 }
перечисление ВидВызывающего (CallerKind) = Линза (Lens) | СессияЦикла (CycleSession) | ХодЧата (ChatTurn) | Вратарь (Gatekeeper)
тип ПределыАгента (AgentLimits) { срок_тишины (silence_term) ?число, бюджет_токенов (token_budget) ?число, бюджет_стоимости (cost_budget) ?дробное, предел_ходов (turn_limit) ?число }
```

## Определения

```
определение есть_критические(находки []Находка) -> да/нет =
  любой(находки, н -> н.критичность == Критичность.Критическая)

определение желательные(находки []Находка) -> []Находка =
  фильтр(находки, н -> н.критичность == Критичность.Желательная)

определение желательные_и_значимые(находки []Находка) -> []Находка =
  фильтр(находки, н -> н.критичность == Критичность.Желательная или н.критичность == Критичность.Значимая)

определение желательные_и_значимые_с_повтором(находки []Находка) -> []Находка =
  фильтр(находки, н -> н.критичность == Критичность.Желательная
                      или (н.критичность == Критичность.Значимая и н.повтор))

определение критические_и_значимые_без_повтора(находки []Находка) -> []Находка =
  фильтр(находки, н -> н.критичность == Критичность.Критическая
                      или (н.критичность == Критичность.Значимая и не н.повтор))

определение не_вошедшие_в(находки []Находка, среди []Находка) -> []Находка =
  фильтр(находки, н -> не любой(среди, с -> с.доказательство == н.доказательство))

определение объединить_без_повторов(а []Находка, б []Находка) -> []Находка =
  а + не_вошедшие_в(б, а)

определение находки_всех_линз(порядок []текст, находки словарь[(текст, число>=0), []Находка], н число>=0) -> []Находка =
  свёртка(порядок, [], (акк, л) -> акк + (находки[(л, н)] ?? []))

определение есть_вопросы_проверяющих(порядок []текст, находки словарь[(текст, число>=0), []Находка], н число>=0) -> да/нет =
  любой(порядок, л -> любой(находки[(л, н)] ?? [], ф -> ф.вопрос))

определение вопросы_проверяющих(порядок []текст, находки словарь[(текст, число>=0), []Находка], н число>=0) -> ?[]Находка =
  пусть вопросы = фильтр(находки_всех_линз(порядок, находки, н), ф -> ф.вопрос) в
  если вопросы == [] то пусто иначе вопросы

определение линзы_без_критических(порядок []текст, находки словарь[(текст, число>=0), []Находка], н число>=0) -> []текст =
  фильтр(порядок, л -> не есть_критические(находки[(л, н)] ?? []))

определение вид_вызова(сторона текст, линзы словарь[текст, Линза]) -> ВидВызывающего =
  если есть(линзы[сторона]) то ВидВызывающего.Линза иначе ВидВызывающего.ХодЧата

определение пределы_роли() -> ПределыАгента =
  ПределыАгента { срок_тишины: если СРОК_БЕЗДЕЙСТВИЯ == 0 то пусто иначе СРОК_БЕЗДЕЙСТВИЯ,
                  бюджет_токенов: пусто, бюджет_стоимости: пусто, предел_ходов: пусто }

определение слово_да_нет(б да/нет) -> текст =
  если б то "yes" иначе "no"
```
