# Машина: Шаг агента

machine: agent_step
machine_version: pilot-2.5
format: 2.22

> Пилот «Шаг агента»: выписка методики по формату 2.1 (повторный пилот выразимости), приведённая к формату 2.5 по правкам формата: нотация объявлений без
> двоеточия (2.3), наборы литералов вместо «текст из» (2.5), столбцы `итог` и `счётчик` «Сторон» (2.2), фрагмент «Пауза» подключением (раздел 9),
> `открытые_запросы` вносит фрагмент. Состав и тексты строк — как в выписке; прозу выписки (источники, правила выписки, топология, расшифровка меток ⟦Нn⟧)
> см. в её файле.

## Функции

```
функция сумма_целых(а ?число, б ?число) -> ?число =>
  если есть(а) и есть(б) то { (а ?? 0) + (б ?? 0) } иначе { а ?? б }

функция сумма_дробных(а ?вещественное, б ?вещественное) -> ?вещественное =>
  если есть(а) и есть(б) то { (а ?? 0.0) + (б ?? 0.0) } иначе { а ?? б }

функция сложить_токены(а Токены, б Токены) -> Токены =>
  Токены { вход: сумма_целых(а.вход, б.вход), выход: сумма_целых(а.выход, б.выход),
           рассуждение: сумма_целых(а.рассуждение, б.рассуждение),
           кэш_чтение: сумма_целых(а.кэш_чтение, б.кэш_чтение), кэш_запись: сумма_целых(а.кэш_запись, б.кэш_запись) }

функция сложить_расход(пред ?РасходТранспорта, нов РасходТранспорта) -> РасходТранспорта =>
  если есть(пред)
  то { ({ пусть п = обязательно(пред);
        РасходТранспорта { модель: нов.модель, токены: сложить_токены(п.токены, нов.токены),
                           стоимость: сумма_дробных(п.стоимость, нов.стоимость),
                           занято_контекста: нов.занято_контекста, начало_запроса: нов.начало_запроса } }) }
  иначе { нов }

функция израсходовано_токенов(р ?РасходТранспорта) -> число =>
  если есть(р)
  то { ({ пусть п = обязательно(р); (п.токены.вход ?? 0) + (п.токены.выход ?? 0) + (п.токены.рассуждение ?? 0) }) }
  иначе { 0 }

функция израсходовано_стоимости(р ?РасходТранспорта) -> вещественное =>
  если есть(р) то { обязательно(р).стоимость ?? 0.0 } иначе { 0.0 }

функция предел_токенов_исчерпан(з ЗапросАгента, р ?РасходТранспорта) -> булево =>
  есть(з.пределы.бюджет_токенов) и израсходовано_токенов(р) >= (з.пределы.бюджет_токенов ?? 0)

функция предел_стоимости_исчерпан(з ЗапросАгента, р ?РасходТранспорта) -> булево =>
  есть(з.пределы.бюджет_стоимости) и израсходовано_стоимости(р) >= (з.пределы.бюджет_стоимости ?? 0.0)

функция предел_ходов_исчерпан(з ЗапросАгента, ходов число>=0) -> булево =>
  ходов >= (з.пределы.предел_ходов ?? ПРЕДЕЛ_ХОДОВ)

функция интерактивный(вид ВидВызывающего) -> булево =>
  вид == ВидВызывающего.СессияЦикла или вид == ВидВызывающего.ХодЧата

функция требует_повышенного_внимания(д ВидДействия) -> булево =>
  выбор д { ВидДействия.Удаление | ВидДействия.Перенос | ВидДействия.Исполнение | ВидДействия.Сеть | ВидДействия.СменаРежима => истина,
            иначе => ложь }

функция решение_по_политике(д ВидДействия, вид ВидВызывающего) -> "allow" | "deny" | "ask" {
  пусть т = (если требует_повышенного_внимания(д) то { "ask" } иначе { ПОЛИТИКА_РАЗРЕШЕНИЙ[д] });
  если т == "ask" и не интерактивный(вид) то { "deny" } иначе { т }
}

функция пауза_повтора(п число>=0) -> число =>
  (1..п).свёртка(ПАУЗА_ПОВТОРА, (н, _) => н * 2)

функция срок_хода(з ЗапросАгента, ход строка, п число>=0) -> число {
  пусть добавка = (если ход == "retry" то { пауза_повтора(п) } иначе { 0 });
  пусть с = з.пределы.срок_тишины;
  если есть(с) то { (с ?? 0) + добавка } иначе { 0 }
}

функция собрать_использование(з ЗапросАгента, ост строка, р ?РасходТранспорта, нач ?число, обн число) -> Использование {
  пусть рез = (если ост == "" то { "reply" } иначе { ост });
  Использование {
    сессия: з.сессия, результат: рез, семья: з.профиль.семья,
    модель: если есть(р) то { обязательно(р).модель }
            иначе { МодельАгента { идентификатор: з.профиль.модель, поставщик: "", вариант: з.профиль.усилие } },
    токены: если есть(р) то { обязательно(р).токены }
            иначе { Токены { вход: пусто, выход: пусто, рассуждение: пусто, кэш_чтение: пусто, кэш_запись: пусто } },
    стоимость: если есть(р) то { обязательно(р).стоимость } иначе { пусто },
    создано: нач ?? обн, обновлено: обн,
    занято_контекста: если есть(р) то { обязательно(р).занято_контекста } иначе { пусто },
    трасса: з.трасса, номер_шага: з.номер_шага }
}

функция событие_расхода(u Использование) -> СобытиеЖурнала =>
  событие_журнала("agent_usage", session: u.сессия, result: u.результат, model: u.модель.идентификатор,
                  input: строка(u.токены.вход), output: строка(u.токены.выход), reasoning: строка(u.токены.рассуждение),
                  cache_read: строка(u.токены.кэш_чтение), cache_write: строка(u.токены.кэш_запись), cost: строка(u.стоимость))
```

### ПОВТОР — развилка

| № | Условие | Действия | Куда |

| 1 | число_повторов < ЧИСЛО_ПОВТОРОВ и ход != "retry" | число_повторов += 1; ход_до_повтора = ход; ход = "retry"; журнал(agent_transient_error, retry: строка(число_повторов)) | БЮДЖЕТ |
| 2 | число_повторов < ЧИСЛО_ПОВТОРОВ и ход == "retry" | число_повторов += 1; журнал(agent_transient_error, retry: строка(число_повторов)) | БЮДЖЕТ |
| 3 | число_повторов >= ЧИСЛО_ПОВТОРОВ | остановка = "no_result"; журнал(agent_transient_error_exhausted, retries: строка(ЧИСЛО_ПОВТОРОВ)) | УЧЁТ |

### ПОЛИТИКА — развилка
Читает: запрос_агента, запрос_разрешения
При входе: пусть запрос = обязательно(запрос_разрешения);
           пусть решение = решение_по_политике(запрос.вид_действия, запрос_агента.вид)

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | решение == "allow" | решение_разрешения = "allow"; ход = "permission"; журнал(permission_auto_allowed, action: строка(запрос.вид_действия)) | БЮДЖЕТ |
| 2 | решение == "deny" | решение_разрешения = "deny"; ход = "permission"; журнал(permission_auto_denied, action: строка(запрос.вид_действия)) | БЮДЖЕТ |

| 4 | иначе | ошибка_машины("fork_no_match", fork: "Policy") | — |
