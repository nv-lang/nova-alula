# Машина: Шаг агента

machine: agent_step
machine_version: pilot-2.5
format: 2.14

> Пилот «Шаг агента»: выписка методики по формату 2.1 (повторный пилот выразимости), приведённая к формату 2.5 по правкам формата: нотация объявлений без
> двоеточия (2.3), наборы литералов вместо «текст из» (2.5), столбцы `итог` и `счётчик` «Сторон» (2.2), фрагмент «Пауза» подключением (раздел 9),
> `открытые_запросы` вносит фрагмент. Состав и тексты строк — как в выписке; прозу выписки (источники, правила выписки, топология, расшифровка меток ⟦Нn⟧)
> см. в её файле.

## Определения

```
определение сумма_целых(а ?число, б ?число) -> ?число =
  если есть(а) и есть(б) то (а ?? 0) + (б ?? 0) иначе а ?? б

определение сумма_дробных(а ?дробное, б ?дробное) -> ?дробное =
  если есть(а) и есть(б) то (а ?? 0.0) + (б ?? 0.0) иначе а ?? б

определение сложить_токены(а Токены, б Токены) -> Токены =
  Токены { вход: сумма_целых(а.вход, б.вход), выход: сумма_целых(а.выход, б.выход),
           рассуждение: сумма_целых(а.рассуждение, б.рассуждение),
           кэш_чтение: сумма_целых(а.кэш_чтение, б.кэш_чтение), кэш_запись: сумма_целых(а.кэш_запись, б.кэш_запись) }

определение сложить_расход(пред ?РасходТранспорта, нов РасходТранспорта) -> РасходТранспорта =
  если есть(пред)
  то (пусть п = обязательно(пред) в
        РасходТранспорта { модель: нов.модель, токены: сложить_токены(п.токены, нов.токены),
                           стоимость: сумма_дробных(п.стоимость, нов.стоимость),
                           занято_контекста: нов.занято_контекста, начало_запроса: нов.начало_запроса })
  иначе нов

определение израсходовано_токенов(р ?РасходТранспорта) -> число =
  если есть(р)
  то (пусть п = обязательно(р) в (п.токены.вход ?? 0) + (п.токены.выход ?? 0) + (п.токены.рассуждение ?? 0))
  иначе 0

определение израсходовано_стоимости(р ?РасходТранспорта) -> дробное =
  если есть(р) то обязательно(р).стоимость ?? 0.0 иначе 0.0

определение предел_токенов_исчерпан(з ЗапросАгента, р ?РасходТранспорта) -> да/нет =
  есть(з.пределы.бюджет_токенов) и израсходовано_токенов(р) >= (з.пределы.бюджет_токенов ?? 0)

определение предел_стоимости_исчерпан(з ЗапросАгента, р ?РасходТранспорта) -> да/нет =
  есть(з.пределы.бюджет_стоимости) и израсходовано_стоимости(р) >= (з.пределы.бюджет_стоимости ?? 0.0)

определение предел_ходов_исчерпан(з ЗапросАгента, ходов число>=0) -> да/нет =
  ходов >= (з.пределы.предел_ходов ?? ПРЕДЕЛ_ХОДОВ)

определение интерактивный(вид ВидВызывающего) -> да/нет =
  вид == ВидВызывающего.СессияЦикла или вид == ВидВызывающего.ХодЧата

определение требует_повышенного_внимания(д ВидДействия) -> да/нет =
  выбор д { ВидДействия.Удаление | ВидДействия.Перенос | ВидДействия.Исполнение | ВидДействия.Сеть | ВидДействия.СменаРежима -> истина;
            иначе -> ложь }

определение решение_по_политике(д ВидДействия, вид ВидВызывающего) -> "allow" | "deny" | "ask" =
  пусть т = (если требует_повышенного_внимания(д) то "ask" иначе ПОЛИТИКА_РАЗРЕШЕНИЙ[д]) в
  если т == "ask" и не интерактивный(вид) то "deny" иначе т

определение пауза_повтора(п число>=0) -> число =

определение срок_хода(з ЗапросАгента, ход текст, п число>=0) -> число =
  пусть добавка = (если ход == "retry" то пауза_повтора(п) иначе 0) в
  пусть с = з.пределы.срок_тишины в
  если есть(с) то (с ?? 0) + добавка иначе 0

определение собрать_использование(з ЗапросАгента, ост текст, р ?РасходТранспорта, нач ?число, обн число) -> Использование =

    занято_контекста: если есть(р) то обязательно(р).занято_контекста иначе пусто,
    трасса: з.трасса, номер_шага: з.номер_шага }
```

### ПОВТОР — развилка (Retry)
Читает: число_повторов, ход

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | число_повторов < ЧИСЛО_ПОВТОРОВ и ход != "retry" | число_повторов += 1; ход_до_повтора = ход; ход = "retry"; журнал(agent_transient_error, retry: текст(число_повторов)) | БЮДЖЕТ |
| 2 | число_повторов < ЧИСЛО_ПОВТОРОВ и ход == "retry" | число_повторов += 1; журнал(agent_transient_error, retry: текст(число_повторов)) | БЮДЖЕТ |
| 3 | число_повторов >= ЧИСЛО_ПОВТОРОВ | остановка = "no_result"; журнал(agent_transient_error_exhausted, retries: текст(ЧИСЛО_ПОВТОРОВ)) | УЧЁТ |
| 4 | иначе | ошибка_машины("fork_no_match", fork: "Retry") | — |

### ПРОДОЛЖЕНИЕ — развилка (Continuation)

### ПРОВЕРКА_ОТВЕТА — развилка (ReplyValidation)
Читает: число_возвратов, текст_ответа, артефакт_ответа, запрос_агента
При входе: пусть проверка = вызвать(reply_validation)

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | проверка == ПроверкаОтвета.Верна | — | УЧЁТ |
| 2 | проверка == ПроверкаОтвета.Пуста | остановка = "no_result"; журнал(agent_no_result) | УЧЁТ |
| 3 | проверка == ПроверкаОтвета.Нарушена и число_возвратов < ЧИСЛО_ВОЗВРАТОВ_ФОРМЫ | число_возвратов += 1; ход = "fix"; журнал(reply_form_violated) | БЮДЖЕТ |
| 4 | проверка == ПроверкаОтвета.Нарушена и число_возвратов >= ЧИСЛО_ВОЗВРАТОВ_ФОРМЫ | остановка = "form_not_fixed"; журнал(reply_form_not_fixed) | УЧЁТ |
| 5 | иначе | ошибка_машины("fork_no_match", fork: "ReplyValidation") | — |

### ПОЛИТИКА — развилка (Policy)
Читает: запрос_агента, запрос_разрешения
При входе: пусть запрос = обязательно(запрос_разрешения);
           пусть решение = решение_по_политике(запрос.вид_действия, запрос_агента.вид)

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | решение == "allow" | решение_разрешения = "allow"; ход = "permission"; журнал(permission_auto_allowed, action: текст(запрос.вид_действия)) | БЮДЖЕТ |
| 2 | решение == "deny" | решение_разрешения = "deny"; ход = "permission"; журнал(permission_auto_denied, action: текст(запрос.вид_действия)) | БЮДЖЕТ |
| 3 | решение == "ask" | — | РАЗРЕШЕНИЕ |
| 4 | иначе | ошибка_машины("fork_no_match", fork: "Policy") | — |

### РАЗРЕШЕНИЕ — обращение (Permission)
Сторона: вратарь
Запрос:

| поле запроса | выражение | читает |
|---|---|---|
| запрос_агента | запрос_агента | сторона |
| запрос_разрешения | запрос_разрешения | сторона |

| № | Событие | Условие | Действия | Куда |
|---|---|---|---|---|
| 1 | сигнал(разрешить) | — | решение_разрешения = "allow"; ход = "permission"; журнал(permission_human_allowed, action: текст(обязательно(запрос_разрешения).вид_действия)) | БЮДЖЕТ |
| 2 | сигнал(отказать) | — | решение_разрешения = "deny"; ход = "permission"; журнал(permission_human_denied, action: текст(обязательно(запрос_разрешения).вид_действия)) | БЮДЖЕТ |

### БЮДЖЕТ — развилка (Budget)
Читает: запрос_агента, расход, число_ходов
При входе: пусть токены_исчерпаны = предел_токенов_исчерпан(запрос_агента, расход);
           пусть стоимость_исчерпана = предел_стоимости_исчерпан(запрос_агента, расход);
           пусть ходы_исчерпаны = предел_ходов_исчерпан(запрос_агента, число_ходов)

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | токены_исчерпаны | остановка = "max_tokens"; журнал(token_limit, spent: текст(израсходовано_токенов(расход))) | УЧЁТ |
| 2 | не токены_исчерпаны и стоимость_исчерпана | остановка = "cost_limit"; журнал(cost_limit, spent: текст(израсходовано_стоимости(расход))) | УЧЁТ |
| 3 | не токены_исчерпаны и не стоимость_исчерпана и ходы_исчерпаны | остановка = "max_turn_requests"; журнал(turn_limit, sent: текст(число_ходов)) | УЧЁТ |
| 4 | не токены_исчерпаны и не стоимость_исчерпана и не ходы_исчерпаны | число_ходов += 1 | ХОД_АГЕНТА |
| 5 | иначе | ошибка_машины("fork_no_match", fork: "Budget") | — |

### УЧЁТ — операция (Accounting)
Читает: запрос_агента, остановка, расход, начато
При входе: пусть сейчас = вызвать(time_now);
           пусть собранное = собрать_использование(запрос_агента, остановка, расход, начато, сейчас)

| № | Событие | Условие | Действия | Куда |
|---|---|---|---|---|
| 1 | сразу | — | использование = собранное; журнал(событие_расхода(собранное)) | ИТОГ_ШАГА |

### ИТОГ_ШАГА — развилка (StepOutcome)
Читает: остановка

| № | Условие | Действия | Куда |
|---|---|---|---|
| 1 | остановка == "" | — | КОНЕЦ |
| 2 | остановка != "" | — | ОСТАНОВЛЕНО |
| 3 | иначе | ошибка_машины("fork_no_match", fork: "StepOutcome") | — |

### КОНЕЦ — конец (End)
Читает: расход, текст_ответа, артефакт_ответа, использование

### ОСТАНОВЛЕНО — конец (Halted)
Читает: использование, запрос_агента, остановка, расход, начато
При входе: пусть сейчас = вызвать(time_now);
           вызвать(close_agent_turn);
           если не есть(использование) то (журнал(событие_расхода(собрать_использование(запрос_агента, остановка, расход, начато, сейчас))));
           журнал(agent_step_stopped, reason: остановка)
Кроме: остановка in {""}

### ОШИБКА — конец (Errored)
Читает: все_данные
При входе: пусть сейчас = вызвать(time_now);
           вызвать(close_agent_turn);
           если не есть(использование) то (журнал(событие_расхода(собрать_использование(запрос_агента, остановка, расход, начато, сейчас))));
           журнал(запись_перехода);
           журнал(все_данные)
